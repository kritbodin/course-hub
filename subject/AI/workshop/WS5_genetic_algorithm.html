📚 ทฤษฎีพื้นฐาน Genetic Algorithm
📝 ความรู้พื้นฐาน: Genetic Algorithm (GA) เป็นเทคนิคการค้นหาและการเพิ่มประสิทธิภาพที่เลียนแบบกระบวนการวิวัฒนาการทางธรรมชาติ ใช้หลักการคัดเลือกธรรมชาติ (Natural Selection) การผสมพันธุ์ (Crossover) และการกลายพันธุ์ (Mutation)
1

5

เปลี่ยนแปลงข้อมูลเล็กน้อย

🔬 การสร้าง Genetic Algorithm แบบสมบูรณ์
📝 โจทย์ตัวอย่างที่ 1: กำหนดประชากร 4 ตัว ดังนี้ 123, 543, 121, 001 กำหนดค่าความเหมาะสมของแต่ละโครโมโซม คือ ผลรวมของตัวเลข ให้ใช้เทคนิค GA คำนวณหาประชากรรุ่นลูกที่มีค่าความเหมาะสมแบบร้อยละที่ดีที่สุดใน 1 รุ่นลูก โดยกำหนดประชากรในรูปแบบฐานสอง การคัดเลือกจากมากไปน้อย การไขว้เปลี่ยนและการกลายพันธุ์สุ่มแบบ 1 จุด
# ===== Genetic Algorithm แบบสมบูรณ์ - ตัวอย่างที่ 1 ===== import random import numpy as np import matplotlib.pyplot as plt class GeneticAlgorithmExample1: def __init__(self, initial_population=[123, 543, 121, 1]): self.initial_population = initial_population self.population_size = len(initial_population) self.chromosome_length = 9 # 3 หลัก x 3 บิตต่อหลัก self.fitness_history = [] def decimal_to_binary_chromosome(self, decimal_number): """แปลงเลขฐาน 10 เป็น binary chromosome""" # เติม 0 ข้างหน้าให้ครบ 3 หลัก padded_number = str(decimal_number).zfill(3) binary_chromosome = "" for digit in padded_number: # แปลงแต่ละหลักเป็น 3 บิต binary_digit = format(int(digit), '03b') binary_chromosome += binary_digit return binary_chromosome def binary_chromosome_to_decimal(self, binary_chromosome): """แปลง binary chromosome เป็นเลขฐาน 10""" # แบ่งเป็นกลุ่มๆ ละ 3 บิต decimal_digits = [] for i in range(0, len(binary_chromosome), 3): binary_group = binary_chromosome[i:i+3] decimal_digit = int(binary_group, 2) decimal_digits.append(str(decimal_digit)) return int(''.join(decimal_digits)) def fitness_function(self, chromosome): """คำนวณค่า fitness (ผลรวมของตัวเลข)""" decimal_number = self.binary_chromosome_to_decimal(chromosome) return sum(int(digit) for digit in str(decimal_number)) def evaluate_population(self, population): """ประเมิน fitness ของประชากรทั้งหมด""" fitness_scores = [] for individual in population: fitness = self.fitness_function(individual) fitness_scores.append(fitness) return fitness_scores def calculate_fitness_percentage(self, fitness_scores): """คำนวณเปอร์เซ็นต์ fitness""" total_fitness = sum(fitness_scores) percentages = [(fitness * 100) / total_fitness for fitness in fitness_scores] return percentages, total_fitness def selection_by_fitness(self, population, fitness_scores): """การคัดเลือกจากมากไปน้อย""" # จับคู่ประชากรกับ fitness paired_data = list(zip(population, fitness_scores)) # เรียงลำดับตาม fitness จากมากไปน้อย paired_data.sort(key=lambda x: x[1], reverse=True) # แยกประชากรที่เรียงแล้ว sorted_population = [individual for individual, fitness in paired_data] sorted_fitness = [fitness for individual, fitness in paired_data] return sorted_population, sorted_fitness def single_point_crossover(self, parent1, parent2, crossover_point): """การผสมพันธุ์แบบ single-point crossover""" # crossover_point นับจากขวาไปซ้าย total_length = len(parent1) cut_position = total_length - crossover_point child1 = parent1[:cut_position] + parent2[cut_position:] child2 = parent2[:cut_position] + parent1[cut_position:] return child1, child2 def bit_flip_mutation(self, chromosome, mutation_group, mutation_position): """การกลายพันธุ์แบบ bit flip""" # mutation_group: กลุ่มที่จะ mutate (นับจากขวาไปซ้าย) # mutation_position: ตำแหน่งในกลุ่ม (นับจากขวาไปซ้าย) mutated = list(chromosome) # หาตำแหน่งที่แท้จริงในสตริง groups = [chromosome[i:i+3] for i in range(0, len(chromosome), 3)] group_index = len(groups) - mutation_group # จากขวาไปซ้าย bit_index_in_group = 3 - mutation_position # จากขวาไปซ้าย overall_position = group_index * 3 + bit_index_in_group # พลิกบิต if overall_position < len(mutated): mutated[overall_position] = '1' if mutated[overall_position] == '0' else '0' return ''.join(mutated) def run_example1(self): """รันตัวอย่างที่ 1 ตามเอกสาร""" print("=== การรัน Genetic Algorithm ตัวอย่างที่ 1 ===") print(f"ประชากรเริ่มต้น: {self.initial_population}") print() # ขั้นตอนที่ 1: แปลงเป็น binary binary_population = [] for individual in self.initial_population: binary_chromosome = self.decimal_to_binary_chromosome(individual) binary_population.append(binary_chromosome) print(f"{individual} -> {binary_chromosome}") print() # ขั้นตอนที่ 2: คำนวณ fitness fitness_scores = self.evaluate_population(binary_population) percentages, total_fitness = self.calculate_fitness_percentage(fitness_scores) print("=== การคำนวณ Fitness ===") for i, (individual, binary, fitness, percentage) in enumerate( zip(self.initial_population, binary_population, fitness_scores, percentages)): print(f"{individual} ({binary}): Fitness = {fitness}, Percentage = {percentage:.2f}%") print(f"Total Fitness: {total_fitness}") print() # ขั้นตอนที่ 3: การคัดเลือก sorted_population, sorted_fitness = self.selection_by_fitness(binary_population, fitness_scores) print("=== การคัดเลือก (จากมากไปน้อย) ===") for i, (chromosome, fitness) in enumerate(zip(sorted_population, sorted_fitness)): decimal = self.binary_chromosome_to_decimal(chromosome) print(f"ลำดับที่ {i+1}: {decimal} ({chromosome}) - Fitness: {fitness}") print() # ขั้นตอนที่ 4: การผสมพันธุ์ (ชุดที่ 2 กับ 1 นับจากขวาไปซ้าย) # ชุดที่ 2 จากขวา = index 2, ชุดที่ 1 จากขวา = index 3 parent1 = sorted_population[2] # 121 parent2 = sorted_population[3] # 001 print("=== การผสมพันธุ์ (Crossover) ===") print(f"Parent 1: {self.binary_chromosome_to_decimal(parent1)} ({parent1})") print(f"Parent 2: {self.binary_chromosome_to_decimal(parent2)} ({parent2})") # Crossover แบบ 1 จุด child1, child2 = self.single_point_crossover(parent1, parent2, crossover_point=1) print(f"Child 1: {self.binary_chromosome_to_decimal(child1)} ({child1})") print(f"Child 2: {self.binary_chromosome_to_decimal(child2)} ({child2})") print() # อัพเดทประชากร new_population = [ sorted_population[0], # 543 เหมือนเดิม sorted_population[1], # 123 เหมือนเดิม child1, child2 ] # ขั้นตอนที่ 5: การกลายพันธุ์ (ชุดที่ 2 จากขวา ตำแหน่งที่ 2) print("=== การกลายพันธุ์ (Mutation) ===") mutation_target = new_population[2] # ชุดที่ 2 จากขวา print(f"ก่อน Mutation: {self.binary_chromosome_to_decimal(mutation_target)} ({mutation_target})") mutated_individual = self.bit_flip_mutation(mutation_target, mutation_group=2, mutation_position=2) print(f"หลัง Mutation: {self.binary_chromosome_to_decimal(mutated_individual)} ({mutated_individual})") print() # ประชากรสุดท้าย final_population = [ new_population[0], new_population[1], mutated_individual, new_population[3] ] print("=== ประชากรรุ่นใหม่ (รุ่นลูก) ===") final_fitness = self.evaluate_population(final_population) final_percentages, final_total = self.calculate_fitness_percentage(final_fitness) for i, (chromosome, fitness, percentage) in enumerate( zip(final_population, final_fitness, final_percentages)): decimal = self.binary_chromosome_to_decimal(chromosome) print(f"ลำดับที่ {i+1}: {decimal} ({chromosome}) - Fitness: {fitness}, Percentage: {percentage:.2f}%") print(f"Total Fitness: {final_total}") # เปรียบเทียบรุ่นเก่าและรุ่นใหม่ print("\n=== การเปรียบเทียบ ===") print(f"รุ่นเดิม - Average Fitness: {np.mean(fitness_scores):.2f}") print(f"รุ่นใหม่ - Average Fitness: {np.mean(final_fitness):.2f}") print(f"การพัฒนา: {np.mean(final_fitness) - np.mean(fitness_scores):.2f}") return final_population, final_fitness # รันตัวอย่างที่ 1 ga_example1 = GeneticAlgorithmExample1() final_population, final_fitness = ga_example1.run_example1()
