📖 ส่วนที่ 1: ทฤษฎี k-Nearest Neighbors
📝 ความรู้พื้นฐาน: k-Nearest Neighbors (kNN) เป็นอัลกอริทึมการจำแนกประเภท (Classification) และการทำนาย (Prediction) ที่ใช้หลักการ "บอกสภาพโดยดูเพื่อนบ้าน" โดยมองหาข้อมูลที่มีความใกล้เคียงกันมากที่สุด

🎯 หลักการทำงานของ kNN

จินตนาการว่าคุณย้ายไปอยู่ย่านใหม่ และต้องการรู้ว่าย่านนี้เป็นอย่างไร

วิธี kNN: ดูบ้าน k หลังที่อยู่ใกล้บ้านคุณมากที่สุด แล้วดูว่าเป็นแบบไหน

  • หาก k=3 และบ้าน 3 หลังใกล้ที่สุดเป็นบ้านหรู → ย่านนี้น่าจะเป็นย่านหรู
  • หาก k=5 และใน 5 หลังใกล้ที่สุด มี 3 หลังเป็นบ้านธรรมดา → ย่านนี้น่าจะเป็นย่านธรรมดา

สูตรระยะทาง Euclidean

d = √(Σ(pi - qi)²)

โดยที่:

  • d = ระยะทางระหว่างจุดสองจุด
  • p = ค่าในชุดข้อมูลที่ต้องการค้นหา
  • q = ค่าในชุดข้อมูลข้างเคียงที่มีอยู่
  • i = ตำแหน่งของแอททริบิวต์ (attribute)
🔍 ขั้นตอนการทำงาน kNN:
  1. คำนวณระยะทาง: หาระยะทางจากจุดที่ต้องการจำแนกไปยังทุกจุดในข้อมูล
  2. เรียงลำดับ: เรียงระยะทางจากน้อยไปมาก
  3. เลือก k จุดใกล้ที่สุด: เอา k จุดแรกที่มีระยะทางน้อยที่สุด
  4. โหวต: ดูว่าใน k จุด กลุ่มไหนมีจำนวนมากที่สุด
  5. ตัดสินใจ: จำแนกให้อยู่ในกลุ่มที่ได้คะแนนโหวตมากที่สุด

# ===== การใช้ kNN สำหรับการจำแนกสัตว์ ===== import numpy as np import pandas as pd from sklearn.neighbors import KNeighborsClassifier from sklearn.metrics import accuracy_score import matplotlib.pyplot as plt # ข้อมูลสัตว์ (ตามตารางข้างต้น) animals_data = { 'สัตว์': ['กบ', 'เป็ด', 'ค้างคาว', 'งู', 'ปลา'], 'ออกลูกเป็นตัว': [0, 0, 1, 0, 0], 'ออกลูกเป็นไข่': [1, 1, 0, 1, 1], 'เลี้ยงลูกด้วยน้ำนม': [0, 0, 1, 0, 0], 'มีขนตามร่างกาย': [0, 1, 1, 0, 0], 'ผิวหยาบหรือมีเกล็ด': [0, 0, 0, 1, 1], 'สัตว์เลือดอุ่น': [0, 1, 0, 0, 0], 'สัตว์เลือดเย็น': [1, 0, 0, 1, 1], 'อยู่ทั้งในน้ำและบนบก': [1, 1, 0, 0, 0], 'หายใจด้วยเหงือก': [0, 0, 0, 0, 1], 'สายพันธุ์': ['สัตว์ครึ่งบกครึ่งน้ำ', 'สัตว์จำพวกนก', 'สัตว์เลี้ยงลูกด้วยนม', 'สัตว์เลื้อยคลาน', 'สัตว์จำพวกปลา'] } # สร้าง DataFrame df = pd.DataFrame(animals_data) print("=== ข้อมูลสัตว์ ===") print(df) # แยกข้อมูล features และ labels feature_columns = ['ออกลูกเป็นตัว', 'ออกลูกเป็นไข่', 'เลี้ยงลูกด้วยน้ำนม', 'มีขนตามร่างกาย', 'ผิวหยาบหรือมีเกล็ด', 'สัตว์เลือดอุ่น', 'สัตว์เลือดเย็น', 'อยู่ทั้งในน้ำและบนบก', 'หายใจด้วยเหงือก'] X = df[feature_columns].values y = df['สายพันธุ์'].values print(f"\nรูปแบบข้อมูล X: {X.shape}") print(f"รูปแบบข้อมูล y: {y.shape}") # สัตว์ปริศนา: ออกลูกเป็นไข่และอยู่ทั้งในน้ำและบนบก mystery_animal = np.array([[0, 1, 0, 0, 0, 0, 0, 1, 0]]) print("\n=== สัตว์ปริศนา ===") print(f"คุณสมบัติ: {mystery_animal}") def manual_knn_classification(X_train, y_train, X_test, k=1): """ การใช้ kNN แบบคำนวณเอง """ distances = [] # คำนวณระยะทาง Euclidean for i, train_sample in enumerate(X_train): # สูตร: d = sqrt(sum((p - q)^2)) distance = np.sqrt(np.sum((X_test - train_sample) ** 2)) distances.append((distance, y_train[i], i)) print(f"ระยะทางถึง {y_train[i]}: {distance:.4f}") # เรียงลำดับตามระยะทาง distances.sort(key=lambda x: x[0]) print(f"\n=== k={k} Nearest Neighbors ===") k_nearest = distances[:k] for i, (dist, label, idx) in enumerate(k_nearest, 1): print(f"{i}. {label} (ระยะทาง: {dist:.4f})") # นับคะแนนโหวต votes = {} for _, label, _ in k_nearest: votes[label] = votes.get(label, 0) + 1 print(f"\nการโหวต: {votes}") # หาผู้ชนะ predicted_class = max(votes, key=votes.get) return predicted_class, k_nearest # ทดสอบด้วย k=1 print("\n" + "="*60) print("=== การจำแนกแบบคำนวณเอง (k=1) ===") prediction_k1, nearest_k1 = manual_knn_classification(X, y, mystery_animal, k=1) print(f"\nผลการจำแนก (k=1): {prediction_k1}") # ทดสอบด้วย k=3 print("\n" + "="*60) print("=== การจำแนกแบบคำนวณเอง (k=3) ===") prediction_k3, nearest_k3 = manual_knn_classification(X, y, mystery_animal, k=3) print(f"\nผลการจำแนก (k=3): {prediction_k3}") # เปรียบเทียบกับ sklearn print("\n" + "="*60) print("=== การเปรียบเทียบกับ scikit-learn ===") for k in [1, 3, 5]: knn = KNeighborsClassifier(n_neighbors=k) knn.fit(X, y) sklearn_prediction = knn.predict(mystery_animal) print(f"k={k}: Manual={prediction_k1 if k==1 else prediction_k3}, sklearn={sklearn_prediction[0]}")

สร้าง DataFrame df_extended = pd.DataFrame(extended_animals_data) print("=== ข้อมูลสัตว์ 8 ชนิด ===") print(df_extended) # แยกข้อมูล features และ labels feature_columns = ['ออกลูกเป็นตัว', 'ออกลูกเป็นไข่', 'เลี้ยงลูกด้วยน้ำนม', 'มีขนตามร่างกาย', 'ผิวหยาบหรือมีเกล็ด', 'สัตว์เลือดอุ่น', 'สัตว์เลือดเย็น', 'อยู่ทั้งในน้ำและบนบก', 'หายใจด้วยเหงือก'] X_extended = df_extended[feature_columns].values y_extended = df_extended['สายพันธุ์'].values # สัตว์ปริศนา: ออกลูกเป็นไข่และอยู่ทั้งในน้ำและบนบก mystery_animal = np.array([[0, 1, 0, 0, 0, 0, 0, 1, 0]]) def detailed_knn_analysis(X_train, y_train, animal_names, X_test, k_values): """ วิเคราะห์ kNN แบบละเอียด """ distances = [] # คำนวณระยะทางทั้งหมด print("=== การคำนวณระยะทาง ===") for i, train_sample in enumerate(X_train): distance = np.sqrt(np.sum((X_test - train_sample) ** 2)) distances.append((distance, y_train[i], animal_names[i])) # แสดงการคำนวณแบบละเอียด diff = X_test.flatten() - train_sample diff_squared = diff ** 2 print(f"{animal_names[i]:<12}: ระยะทาง = √({'+'.join(map(str, diff_squared.astype(int)))}) = √{np.sum(diff_squared):.0f} = {distance:.4f}") # เรียงลำดับตามระยะทาง distances.sort(key=lambda x: x[0]) print(f"\n=== การเรียงลำดับตามระยะทาง ===") for i, (dist, label, name) in enumerate(distances, 1): print(f"{i}. {name:<12} ({label:<25}) = {dist:.4f}") # ทดสอบกับค่า k ต่างๆ results = {} for k in k_values: print(f"\n{'='*60}") print(f"=== การวิเคราะห์ k={k} ===") k_nearest = distances[:k] print(f"k={k} เพื่อนบ้านที่ใกล้ที่สุด:") votes = Counter() for i, (dist, label, name) in enumerate(k_nearest, 1): votes[label] += 1 print(f" {i}. {name:<12} ({label:<25}) ระยะทาง: {dist:.4f}") print(f"\nการนับคะแนนโหวต:") for species, count in votes.most_common(): percentage = (count / k) * 100 print(f" {species:<25}: {count}/{k} ({percentage:.1f}%)") # ทำการวิเคราะห์ animal_names = df_extended['สัตว์'].tolist() k_values = [3, 4] print("สัตว์ปริศนา: ออกลูกเป็นไข่และอยู่ทั้งในน้ำและบนบก") print("คุณสมบัติ:", mystery_animal.flatten()) print() results = detailed_knn_analysis(X_extended, y_extended, animal_names, mystery_animal, k_values) # # ===== การปรับปรุง kNN และการสำรวจพารามิเตอร์ ===== import numpy as np import pandas as pd import matplotlib.pyplot as plt from sklearn.neighbors import KNeighborsClassifier from sklearn.model_selection import cross_val_score from sklearn.preprocessing import StandardScaler from sklearn.metrics import confusion_matrix, classification_report import seaborn as sns # ฟังก์ชันสำหรับทดสอบค่า k ต่างๆ def find_optimal_k(X, y, k_range=range(1, 11)): """ หาค่า k ที่เหมาะสมที่สุด """ accuracies = [] for k in k_range: knn = KNeighborsClassifier(n_neighbors=k) # ใช้ cross-validation เพื่อประเมินประสิทธิภาพ scores = cross_val_score(knn, X, y, cv=3, scoring='accuracy') accuracies.append(scores.mean()) print(f"k={k}: ความแม่นยำเฉลี่ย = {scores.mean():.4f} (±{scores.std():.4f})") # หาค่า k ที่ให้ความแม่นยำสูงสุด optimal_k = k_range[np.argmax(accuracies)] max_accuracy = max(accuracies) # สร้างกราฟ plt.figure(figsize=(10, 6)) plt.plot(k_range, accuracies, 'bo-', linewidth=2, markersize=8) plt.axvline(x=optimal_k, color='r', linestyle='--', alpha=0.7) plt.xlabel('ค่า k') plt.ylabel('ความแม่นยำ (Accuracy)') plt.title('การเปรียบเทียบประสิทธิภาพของ kNN ด้วยค่า k ต่างๆ') plt.grid(True, alpha=0.3) plt.xticks(k_range) # เพิ่มข้อความแสดงค่าที่เหมาะสม plt.annotate(f'k ที่เหมาะสม = {optimal_k}\nความแม่นยำ = {max_accuracy:.4f}', xy=(optimal_k, max_accuracy), xytext=(optimal_k+1, max_accuracy-0.02), arrowprops=dict(arrowstyle='->', color='red'), fontsize=12, ha='left') plt.tight_layout() plt.show() return optimal_k, max_accuracy # ฟังก์ชันสำหรับการทดสอบ distance metrics ต่างๆ def compare_distance_metrics(X, y, mystery_sample): """ เปรียบเทียบ distance metrics ต่างๆ """ metrics = { 'euclidean': 'Euclidean (L2)', 'manhattan': 'Manhattan (L1)', 'minkowski': 'Minkowski (p=3)', 'chebyshev': 'Chebyshev' } results = {} print("=== การเปรียบเทียบ Distance Metrics ===") for metric, name in metrics.items(): if metric == 'minkowski': knn = KNeighborsClassifier(n_neighbors=3, metric=metric, p=3) else: knn = KNeighborsClassifier(n_neighbors=3, metric=metric) knn.fit(X, y) prediction = knn.predict(mystery_sample) # หา neighbors และระยะทาง distances, indices = knn.kneighbors(mystery_sample) print(f"\n{name}:") print(f" ผลการจำแนก: {prediction[0]}") print(f" 3 เพื่อนบ้านใกล้ที่สุด:") for i, (idx, dist) in enumerate(zip(indices[0], distances[0])): print(f" {i+1}. {df_extended.iloc[idx]['สัตว์']} (ระยะทาง: {dist:.4f})") results[name] = { 'prediction': prediction[0], 'distances': distances[0], 'neighbors': [df_extended.iloc[idx]['สัตว์'] for idx in indices[0]] } return results # ฟังก์ชันสำหรับ weighted kNN def weighted_knn_prediction(X, y, mystery_sample, k=3): """ kNN แบบถ่วงน้ำหนักด้วยระยะทาง """ from sklearn.neighbors import KNeighborsClassifier # kNN ปกติ knn_uniform = KNeighborsClassifier(n_neighbors=k, weights='uniform') knn_uniform.fit(X, y) prediction_uniform = knn_uniform.predict(mystery_sample) # kNN แบบถ่วงน้ำหนัก knn_weighted = KNeighborsClassifier(n_neighbors=k, weights='distance') knn_weighted.fit(X, y) prediction_weighted = knn_weighted.predict(mystery_sample) # แสดงผลลัพธ์ print("=== การเปรียบเทียบ kNN ปกติ vs kNN ถ่วงน้ำหนัก ===") print(f"kNN ปกติ (uniform weights): {prediction_uniform[0]}") print(f"kNN ถ่วงน้ำหนัก (distance weights): {prediction_weighted[0]}") # แสดงรายละเอียดเพิ่มเติม distances, indices = knn_uniform.kneighbors(mystery_sample) weights = 1 / (distances[0] + 1e-6) # เพิ่มค่าเล็กๆ เพื่อไม่ให้หารด้วย 0 weights = weights / np.sum(weights) # normalize print(f"\nรายละเอียด k={k} เพื่อนบ้าน:") print(f"{'อันดับ':<6} {'สัตว์':<12} {'สายพันธุ์':<25} {'ระยะทาง':<10} {'น้ำหนัก':<10}") print("-" * 75) for i, (idx, dist, weight) in enumerate(zip(indices[0], distances[0], weights)): animal_name = df_extended.iloc[idx]['สัตว์'] species = df_extended.iloc[idx]['สายพันธุ์'] print(f"{i+1:<6} {animal_name:<12} {species:<25} {dist:<10.4f} {weight:<10.4f}") return prediction_uniform[0], prediction_weighted[0] # ใช้ข้อมูลจากส่วนที่แล้ว X_extended = df_extended[feature_columns].values y_extended = df_extended['สายพันธุ์'].values mystery_animal = np.array([[0, 1, 0, 0, 0, 0, 0, 1, 0]]) print("=== การปรับปรุงประสิทธิภาพ kNN ===") print("ข้อมูล: สัตว์ 8 ชนิด") print("สัตว์ปริศนา: ออกลูกเป็นไข่และอยู่ทั้งในน้ำและบนบก") print() # หาค่า k ที่เหมาะสม print("1. การหาค่า k ที่เหมาะสม") optimal_k, max_acc = find_optimal_k(X_extended, y_extended, range(1, 8)) print(f"ค่า k ที่เหมาะสม: {optimal_k}") print("\n" + "="*60) # เปรียบเทียบ distance metrics print("2. การเปรียบเทียบ Distance Metrics") distance_results = compare_distance_metrics(X_extended, y_extended, mystery_animal) print("\n" + "="*60) # ทดสอบ weighted kNN print("3. การเปรียบเทียบ kNN ปกติ vs ถ่วงน้ำหนัก") uniform_pred, weighted_pred = weighted_knn_prediction(X_extended, y_extended, mystery_animal, k=3) # สรุปผลการทดลอง print("\n" + "="*60) print("=== สรุปผลการทดลอง ===") print(f"ค่า k ที่เหมาะสม: {optimal_k}") print(f"Distance metric ที่ดีที่สุด: Euclidean (มาตรฐาน)") print(f"kNN ปกติ vs ถ่วงน้ำหนัก: {uniform_pred} vs {weighted_pred}") # # ===== ระบบแนะนำภาพยนตร์ด้วย kNN ===== import numpy as np import pandas as pd import matplotlib.pyplot as plt from sklearn.neighbors import KNeighborsClassifier, NearestNeighbors from sklearn.preprocessing import StandardScaler from sklearn.metrics.pairwise import cosine_similarity class MovieRecommendationSystem: def __init__(self): self.scaler = StandardScaler() self.knn_model = NearestNeighbors(n_neighbors=5, metric='cosine') def create_movie_dataset(self): """สร้างข้อมูลภาพยนตร์จำลอง""" movies_data = { 'ภาพยนตร์': [ 'Avengers: Endgame', 'Titanic', 'The Dark Knight', 'Frozen', 'Joker', 'Avatar', 'Toy Story 4', 'The Lion King', 'Spider-Man', 'Interstellar', 'The Notebook', 'Inception', 'Finding Nemo', 'Black Panther', 'La La Land' ], 'แอคชัน': [10, 2, 9, 1, 6, 8, 2, 3, 9, 3, 1, 7, 2, 9, 2], 'โรแมนติก': [2, 9, 1, 3, 2, 3, 1, 4, 3, 2, 10, 2, 2, 3, 9], 'ตลก': [7, 3, 2, 8, 3, 2, 9, 7, 6, 1, 4, 2, 8, 5, 6], 'ดราม่า': [6, 8, 7, 6, 9, 4, 4, 8, 5, 9, 9, 6, 4, 7, 8], 'ไซไฟ': [9, 1, 3, 1, 2, 10, 1, 1, 7, 10, 1, 9, 1, 8, 1], 'แฟนตาซี': [8, 2, 4, 9, 3, 6, 2, 8, 5, 2, 2, 6, 3, 6, 3], 'เด็ก': [5, 2, 2, 10, 2, 3, 10, 9, 6, 2, 2, 2, 9, 4, 3], 'คะแนน_IMDB': [8.4, 7.8, 9.0, 7.4, 8.4, 7.8, 7.7, 6.8, 7.3, 8.6, 7.8, 8.8, 8.2, 7.3, 8.0], 'ปีที่ออก': [2019, 1997, 2008, 2013, 2019, 2009, 2019, 2019, 2002, 2014, 2004, 2010, 2003, 2018, 2016] } return pd.DataFrame(movies_data) def fit(self, df): """ฝึกฝนโมเดล""" # เลือก features สำหรับการคำนวณความคล้าย self.feature_cols = ['แอคชัน', 'โรแมนติก', 'ตลก', 'ดราม่า', 'ไซไฟ', 'แฟนตาซี', 'เด็ก', 'คะแนน_IMDB'] self.df = df.copy() # Normalize ข้อมูล self.features_scaled = self.scaler.fit_transform(df[self.feature_cols]) # ฝึกฝน kNN self.knn_model.fit(self.features_scaled) print("✅ ระบบแนะนำพร้อมใช้งาน!") def get_recommendations(self, movie_name, n_recommendations=5): """แนะนำภาพยนตร์คล้ายกัน""" try: # หา index ของภาพยนตร์ movie_idx = self.df[self.df['ภาพยนตร์'] == movie_name].index[0] # หาภาพยนตร์ที่คล้ายกัน distances, indices = self.knn_model.kneighbors([self.features_scaled[movie_idx]], n_neighbors=n_recommendations+1) # ลบภาพยนตร์ต้นฉบับออก (อันดับแรกจะเป็นตัวเอง) similar_indices = indices[0][1:] similar_distances = distances[0][1:] print(f"🎬 ภาพยนตร์ที่คล้ายกับ '{movie_name}':") print("-" * 80) recommendations = [] for i, (idx, distance) in enumerate(zip(similar_indices, similar_distances)): similar_movie = self.df.iloc[idx] similarity = 1 - distance # แปลง distance เป็น similarity recommendations.append({ 'ภาพยนตร์': similar_movie['ภาพยนตร์'], 'ความคล้าย': similarity, 'คะแนน_IMDB': similar_movie['คะแนน_IMDB'], 'ปี': similar_movie['ปีที่ออก'] }) print(f"{i+1}. {similar_movie['ภาพยนตร์']:<20} " f"(ความคล้าย: {similarity:.3f}, " f"IMDB: {similar_movie['คะแนน_IMDB']}, " f"ปี: {similar_movie['ปีที่ออก']})") return recommendations except IndexError: print(f"❌ ไม่พบภาพยนตร์ '{movie_name}' ในระบบ") return [] def analyze_user_profile(self, favorite_movies): """วิเคราะห์รสนิยมผู้ใช้""" user_features = np.zeros(len(self.feature_cols)) valid_movies = [] print("🔍 วิเคราะห์รสนิยมจากภาพยนตร์ที่ชอบ:") print("-" * 50) for movie in favorite_movies: try: movie_data = self.df[self.df['ภาพยนตร์'] == movie].iloc[0] movie_features = movie_data[self.feature_cols].values user_features += movie_features valid_movies.append(movie) print(f"✅ {movie}") except IndexError: print(f"❌ ไม่พบ: {movie}") if len(valid_movies) == 0: print("ไม่มีภาพยนตร์ที่ถูกต้อง") return None # คำนวณค่าเฉลี่ย user_features = user_features / len(valid_movies) print(f"\n📊 โปรไฟล์รสนิยม (คะแนน 1-10):") for feature, score in zip(self.feature_cols, user_features): print(f" {feature:<15}: {score:.1f}") # หาภาพยนตร์ที่เหมาะสม user_features_scaled = self.scaler.transform([user_features]) distances, indices = self.knn_model.kneighbors(user_features_scaled, n_neighbors=5) print(f"\n🎯 ภาพยนตร์ที่แนะนำตามรสนิยม:") print("-" * 50) recommendations = [] for i, idx in enumerate(indices[0]): if self.df.iloc[idx]['ภาพยนตร์'] not in favorite_movies: movie = self.df.iloc[idx] similarity = 1 - distances[0][i] recommendations.append({ 'ภาพยนตร์': movie['ภาพยนตร์'], 'ความเหมาะสม': similarity, 'คะแนน_IMDB': movie['คะแนน_IMDB'] }) print(f"{len(recommendations)}. {movie['ภาพยนตร์']:<20} " f"(เหมาะสม: {similarity:.3f}, " f"IMDB: {movie['คะแนน_IMDB']})") return recommendations def create_similarity_heatmap(self): """สร้าง heatmap แสดงความคล้ายระหว่างภาพยนตร์""" # คำนวณ cosine similarity similarity_matrix = cosine_similarity(self.features_scaled) plt.figure(figsize=(12, 10)) # สร้าง heatmap import seaborn as sns mask = np.triu(similarity_matrix, k=1) # ซ่อนส่วนบน sns.heatmap(similarity_matrix, annot=False, mask=mask, xticklabels=self.df['ภาพยนตร์'], yticklabels=self.df['ภาพยนตร์'], cmap='viridis', cbar_kws={'label': 'Cosine Similarity'}) plt.title('ความคล้ายระหว่างภาพยนตร์', fontsize=16) plt.xticks(rotation=45, ha='right') plt.yticks(rotation=0) plt.tight_layout() plt.show() # สร้างและทดสอบระบบ recommender = MovieRecommendationSystem() # สร้างข้อมูลภาพยนตร์ movies_df = recommender.create_movie_dataset() print("=== ข้อมูลภาพยนตร์ ===") print(movies_df[['ภาพยนตร์', 'แอคชัน', 'โรแมนติก', 'ตลก', 'คะแนน_IMDB', 'ปีที่ออก']].head(10)) # ฝึกฝนโมเดล print("\n=== การฝึกฝนโมเดล ===") recommender.fit(movies_df) # ทดสอบการแนะนำ print("\n=== การทดสอบระบบแนะนำ ===") # ทดสอบ 1: หาภาพยนตร์คล้าย The Dark Knight recommendations_1 = recommender.get_recommendations('The Dark Knight', n_recommendations=4) print("\n" + "="*80) # ทดสอบ 2: หาภาพยนตร์คล้าย Frozen recommendations_2 = recommender.get_recommendations('Frozen', n_recommendations=4) print("\n" + "="*80) # ทดสอบ 3: วิเคราะห์รสนิยมผู้ใช้ print("=== การวิเคราะห์รสนิยมผู้ใช้ ===") user_favorites = ['Avengers: Endgame', 'The Dark Knight', 'Spider-Man'] user_recommendations = recommender.analyze_user_profile(user_favorites) # สร้าง heatmap print("\n=== การแสดงความคล้ายระหว่างภาพยนตร์ ===") recommender.create_similarity_heatmap()

  • พอใช้ (C): เข้าใจเบื้องต้น ทำงานได้บางส่วน
  • ต้องปรับปรุง (D/F): ไม่เข้าใจทฤษฎี โค้ดไม่ทำงาน
🎯 ส่วนที่ 7: สรุปและแนวทางการประยุกต์
🧠 จุดสำคัญที่ควรจำ:
  1. k-Nearest Neighbors ใช้หลักการ "บอกสภาพโดยดูเพื่อนบ้าน"
  2. ระยะทาง Euclidean เป็น distance metric พื้นฐานที่นิยมใช้
  3. การเลือกค่า k ส่งผลต่อประสิทธิภาพมาก
  4. Data preprocessing มีความสำคัญต่อผลลัพธ์
  5. Lazy learning ไม่ต้องฝึกฝนโมเดลล่วงหน้า
สาขาการประยุกต์ ตัวอย่างการใช้งาน ประโยชน์ E-commerce ระบบแนะนำสินค้า, ลูกค้าที่คล้ายกัน เพิ่มยอดขาย, ความพึงพอใจลูกค้า การแพทย์ การวินิจฉัยโรค, การจับคู่ผู้ป่วย ช่วยแพทย์ตัดสินใจ, หาการรักษา การเงิน การประเมินความเสี่ยงเครดิต ลดความเสี่ยงในการให้สินเชื่อ Image Recognition การจำแนกภาพ, Face recognition ระบบรักษาความปลอดภัย NLP Text classification, Sentiment analysis วิเคราะห์ความรู้สึกจากข้อความ

🚀 โจทย์ท้าทายสำหรับผู้ที่สนใจ:

  1. Music Recommendation: สร้างระบบแนะนำเพลงตาม genre และ mood
  2. House Price Prediction: ทำนายราคาบ้านจากบ้านใกล้เคียง
  3. Fraud Detection: ตรวจจับการทำธุรกรรมผิดปกติ
  4. Gene Classification: จำแนกยีนตามหน้าที่และโครงสร้าง
📚 แนวทางพัฒนาต่อ:
  • 🔗 เรียนรู้ Advanced kNN variants (Weighted kNN, Fuzzy kNN)
  • 📊 ศึกษา Dimensionality Reduction (PCA, t-SNE)
  • ⚡ เรียนรู้ Fast similarity search (LSH, Approximate kNN)
  • 🤖 รวม kNN เข้ากับ Deep Learning
  • 📱 สร้าง Mobile App ที่ใช้ kNN
