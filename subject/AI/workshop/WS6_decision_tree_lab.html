ใบงานฝึกปฏิบัติบทที่ 6: การเรียนรู้ (Learning)

เรื่อง: ต้นไม้ตัดสินใจ (Decision Tree)

📖 ส่วนที่ 1: ทฤษฎีต้นไม้ตัดสินใจ
📝 ความรู้พื้นฐาน: ต้นไม้ตัดสินใจ (Decision Tree) เป็นเทคนิคการเรียนรู้แบบมีผู้ดูแล (Supervised Learning) ที่ใช้สำหรับการจำแนกประเภท (Classification) และการทำนาย (Prediction)

สูตรสำคัญในการสร้างต้นไม้ตัดสินใจ

1. Entropy (เอนโทรปี)

E(S) = -Σ P(vi) × log₂P(vi)

โดยที่ P(vi) = ความน่าจะเป็นของแต่ละคลาส

2. Information Gain (การได้รับข้อมูล)

Gain(S,A) = E(S) - Σ (|Sv|/|S|) × E(Sv)

โดยที่ A = attribute ที่พิจารณา, Sv = subset ของ S ที่มีค่า v ใน attribute A

💡 หลักการทำงาน:
  1. คำนวณ Information Gain ของ attribute ทั้งหมด
  2. เลือก attribute ที่ให้ Information Gain สูงสุดเป็น root node
  3. แบ่งข้อมูลตาม attribute ที่เลือก
  4. ทำซ้ำกับ subset ย่อยจนกว่าจะได้ leaf node
🍰 ส่วนที่ 2: ตัวอย่างการสำรวจลูกค้าสั่งของหวาน
📝 โจทย์: จากข้อมูลการสำรวจลูกค้าที่สั่งของหวาน ให้สร้างต้นไม้ตัดสินใจเพื่อทำนายว่าลูกค้าจะชอบหรือไม่ชอบ
ลูกค้า สั่งของหวาน ของหวาน ความชอบ
A ไม่สั่ง ไอศครีม ไม่ชอบ
B สั่ง ไอศครีม ชอบ
C ไม่สั่ง ขนมเค้ก ไม่ชอบ
D สั่ง ไอศครีม ชอบ
E สั่ง ขนมเค้ก ชอบ
# ===== การคำนวณต้นไม้ตัดสินใจด้วยมือ ===== import pandas as pd import numpy as np import math # ข้อมูลตัวอย่าง data = { 'ลูกค้า': ['A', 'B', 'C', 'D', 'E'], 'สั่งของหวาน': ['ไม่สั่ง', 'สั่ง', 'ไม่สั่ง', 'สั่ง', 'สั่ง'], 'ของหวาน': ['ไอศครีม', 'ไอศครีม', 'ขนมเค้ก', 'ไอศครีม', 'ขนมเค้ก'], 'ความชอบ': ['ไม่ชอบ', 'ชอบ', 'ไม่ชอบ', 'ชอบ', 'ชอบ'] } df = pd.DataFrame(data) print("=== ข้อมูลตัวอย่าง ===") print(df) print() def calculate_entropy(target_col): """คำนวณ Entropy ของคอลัมน์เป้าหมาย""" elements, counts = np.unique(target_col, return_counts=True) entropy = 0 total = len(target_col) print(f"การคำนวณ Entropy:") print(f"ข้อมูลทั้งหมด: {total} รายการ") for i in range(len(elements)): probability = counts[i] / total if probability > 0: entropy_part = -probability * math.log2(probability) entropy += entropy_part print(f" {elements[i]}: {counts[i]} รายการ, P = {probability:.3f}, -P×log₂P = {entropy_part:.4f}") print(f"Entropy รวม = {entropy:.4f}") return entropy def calculate_information_gain(data, split_attribute, target_name): """คำนวณ Information Gain ของ attribute""" print(f"\n=== คำนวณ Information Gain สำหรับ '{split_attribute}' ===") # คำนวณ Entropy ของข้อมูลทั้งหมด total_entropy = calculate_entropy(data[target_name]) # หาค่าที่ไม่ซ้ำใน attribute vals, counts = np.unique(data[split_attribute], return_counts=True) # คำนวณ Weighted Entropy weighted_entropy = 0 total_size = len(data) print(f"\nการคำนวณ Weighted Entropy:") for i in range(len(vals)): val_prob = counts[i] / total_size subset = data[data[split_attribute] == vals[i]] subset_entropy = calculate_entropy(subset[target_name]) weighted_entropy += val_prob * subset_entropy print(f" {vals[i]}: {counts[i]} รายการ, น้ำหนัก = {val_prob:.3f}, Entropy = {subset_entropy:.4f}") print(f"Weighted Entropy รวม = {weighted_entropy:.4f}") # คำนวณ Information Gain information_gain = total_entropy - weighted_entropy print(f"Information Gain = {total_entropy:.4f} - {weighted_entropy:.4f} = {information_gain:.4f}") return information_gain # คำนวณ Information Gain สำหรับแต่ละ attribute print("="*60) gain_ordering = calculate_information_gain(df, 'สั่งของหวาน', 'ความชอบ') print("\n" + "="*60) gain_dessert = calculate_information_gain(df, 'ของหวาน', 'ความชอบ') print("\n=== สรุปผลการคำนวณ ===") print(f"Information Gain (สั่งของหวาน): {gain_ordering:.4f}") print(f"Information Gain (ของหวาน): {gain_dessert:.4f}") if gain_ordering > gain_dessert: print(f"\n✅ เลือก 'สั่งของหวาน' เป็น root node เพราะมี Information Gain สูงสุด") root_attribute = 'สั่งของหวาน' else: print(f"\n✅ เลือก 'ของหวาน' เป็น root node เพราะมี Information Gain สูงสุด") root_attribute = 'ของหวาน'

💡 แบบฝึกหัด 2

ให้ศึกษาการคำนวณ Information Gain ของ attribute "สั่งของหวาน" ตามขั้นตอน:

  1. คำนวณ Entropy ของข้อมูลทั้งหมด E(S)
  2. แบ่งข้อมูลตาม attribute "สั่งของหวาน"
  3. คำนวณ Entropy ของแต่ละ subset
  4. คำนวณ Information Gain

ให้ปรับการเขียนโค้ด Python เพื่อคำนวณตามโจทย์ดังนี้

  1. เปลี่ยน B เป็นไม่สั่ง และไม่ชอบ
  2. เปลี่ยน E เป็นไม่สั่ง และไม่ชอบ
🎯 ส่วนที่ 3: โจทย์ท้าทาย - การวิเคราะห์ลูกค้าร้านอาหาร
📝 สถานการณ์: ร้านอาหารต้องการวิเคราะห์พฤติกรรมการรอของลูกค้า เพื่อปรับปรุงการให้บริการ
ลูกค้า ราคา ร้านอาหาร เวลารอ (นาที) สถานะ
A Expensive Japanese 10-30 Wait
B Cheap Thai 10-30 No
C Expensive Thai 10-30 No
D Expensive Japanese >30 No
E Expensive Japanese 0-10 Wait
F Cheap Thai 0-10 Wait
# ===== โจทย์ท้าทาย: การวิเคราะห์ลูกค้าร้านอาหาร ===== import pandas as pd import numpy as np from sklearn.tree import DecisionTreeClassifier, plot_tree, export_text from sklearn.preprocessing import LabelEncoder import matplotlib.pyplot as plt # ข้อมูลร้านอาหาร restaurant_data = { 'ลูกค้า': ['A', 'B', 'C', 'D', 'E', 'F'], 'ราคา': ['Expensive', 'Cheap', 'Expensive', 'Expensive', 'Expensive', 'Cheap'], 'ร้านอาหาร': ['Japanese', 'Thai', 'Thai', 'Japanese', 'Japanese', 'Thai'], 'เวลารอ': ['10-30', '10-30', '10-30', '>30', '0-10', '0-10'], 'สถานะ': ['Wait', 'No', 'No', 'No', 'Wait', 'Wait'] } df_restaurant = pd.DataFrame(restaurant_data) print("=== ข้อมูลลูกค้าร้านอาหาร ===") print(df_restaurant) # เตรียม Label Encoders le_price = LabelEncoder() le_restaurant = LabelEncoder() le_time = LabelEncoder() le_status = LabelEncoder() # Encode ข้อมูล df_encoded = df_restaurant.copy() df_encoded['ราคา'] = le_price.fit_transform(df_restaurant['ราคา']) df_encoded['ร้านอาหาร'] = le_restaurant.fit_transform(df_restaurant['ร้านอาหาร']) df_encoded['เวลารอ'] = le_time.fit_transform(df_restaurant['เวลารอ']) df_encoded['สถานะ'] = le_status.fit_transform(df_restaurant['สถานะ']) print(f"\n=== การ Encoding ===") print(f"ราคา: {dict(zip(le_price.classes_, range(len(le_price.classes_))))}") print(f"ร้านอาหาร: {dict(zip(le_restaurant.classes_, range(len(le_restaurant.classes_))))}") print(f"เวลารอ: {dict(zip(le_time.classes_, range(len(le_time.classes_))))}") print(f"สถานะ: {dict(zip(le_status.classes_, range(len(le_status.classes_))))}") # เตรียมข้อมูลสำหรับ training X = df_encoded[['ราคา', 'ร้านอาหาร', 'เวลารอ']] y = df_encoded['สถานะ'] # สร้างโมเดล clf_restaurant = DecisionTreeClassifier(criterion='entropy', random_state=42) clf_restaurant.fit(X, y) # แสดงข้อมูลต้นไม้ print(f"\n=== ข้อมูลต้นไม้ ===") print(f"จำนวน node: {clf_restaurant.tree_.node_count}") print(f"ความลึก: {clf_restaurant.tree_.max_depth}") # แสดงกฎในรูปแบบข้อความ feature_names = ['ราคา', 'ร้านอาหาร', 'เวลารอ'] class_names = le_status.classes_ print(f"\n=== กฎของต้นไม้ ===") tree_rules = export_text(clf_restaurant, feature_names=feature_names) print(tree_rules) # วาดต้นไม้ plt.figure(figsize=(15, 10)) plot_tree(clf_restaurant, feature_names=feature_names, class_names=class_names, filled=True, rounded=True, fontsize=10) plt.title('Decision Tree สำหรับการทำนายพฤติกรรมการรอของลูกค้า', fontsize=14, pad=20) plt.show() # การทำนายและการประเมินผล predictions = clf_restaurant.predict(X) accuracy = np.mean(predictions == y) * 100 print(f"\n=== การประเมินผล ===") print(f"ความแม่นยำ: {accuracy:.1f}%") print(f"\nการเปรียบเทียบการทำนายกับค่าจริง:") for i in range(len(df_restaurant)): predicted_encoded = predictions[i] actual_encoded = y.iloc[i] predicted = le_status.inverse_transform([predicted_encoded])[0] actual = le_status.inverse_transform([actual_encoded])[0] print(f"ลูกค้า {df_restaurant.iloc[i]['ลูกค้า']}: " f"ทำนาย={predicted}, จริง={actual} " f"{'✓' if predicted == actual else '✗'}") # ทดสอบกับข้อมูลใหม่ print(f"\n=== การทดสอบกับข้อมูลใหม่ ===") new_customers = [ {'ราคา': 'Cheap', 'ร้านอาหาร': 'Japanese', 'เวลารอ': '0-10'}, {'ราคา': 'Expensive', 'ร้านอาหาร': 'Thai', 'เวลารอ': '>30'}, {'ราคา': 'Cheap', 'ร้านอาหาร': 'Thai', 'เวลารอ': '>30'} ] for i, customer in enumerate(new_customers, 1): # Encode ข้อมูลใหม่ price_enc = le_price.transform([customer['ราคา']])[0] restaurant_enc = le_restaurant.transform([customer['ร้านอาหาร']])[0] time_enc = le_time.transform([customer['เวลารอ']])[0] # ทำนาย prediction_enc = clf_restaurant.predict([[price_enc, restaurant_enc, time_enc]])[0] prediction = le_status.inverse_transform([prediction_enc])[0] # ความน่าจะเป็น probabilities = clf_restaurant.predict_proba([[price_enc, restaurant_enc, time_enc]])[0] prob_dict = dict(zip(class_names, probabilities)) print(f"ลูกค้าใหม่ {i}: {customer}") print(f" การทำนาย: {prediction}") print(f" ความน่าจะเป็น: {prob_dict}") print() # การวิเคราะห์ความสำคัญของ features feature_importance = clf_restaurant.feature_importances_ print(f"=== ความสำคัญของ Features ===") for i, importance in enumerate(feature_importance): print(f"{feature_names[i]}: {importance:.4f}") # สร้างกราฟแสดงความสำคัญ plt.figure(figsize=(10, 6)) plt.bar(feature_names, feature_importance) plt.title('ความสำคัญของ Features ในการตัดสินใจ') plt.ylabel('Feature Importance') plt.show()
