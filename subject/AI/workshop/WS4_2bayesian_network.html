ใบงานฝึกปฏิบัติบทที่ 4: การแก้ไขปัญหา (Problem Solving)

เรื่อง: เบย์เซียนเน็ตเวิร์ค (Bayesian Network)

📖 ส่วนที่ 1: ทบทวนเบย์เซียนเน็ตเวิร์ค
📝 ความรู้พื้นฐาน: เบย์เซียนเน็ตเวิร์คเป็นแบบจำลองกราฟที่แสดงความสัมพันธ์เชิงสาเหตุระหว่างตัวแปรต่างๆ โดยใช้ทฤษฎีของเบย์ในการคำนวณความน่าจะเป็น

สูตรความน่าจะเป็นร่วมกัน (Joint Probability)

P(x₁, x₂, ..., xₙ) = ∏ᵢ₌₁ⁿ P(Xᵢ|Parents(Xᵢ))

โดยที่:

  • P(Xᵢ|Parents(Xᵢ)) = ความน่าจะเป็นของตัวแปร Xᵢ เมื่อรู้ค่าของโหนดพ่อแม่
  • Parents(Xᵢ) = เซตของโหนดพ่อแม่ของ Xᵢ
  • ∏ = การคูณทุกค่า (Product)
🌐 ส่วนที่ 2: ปัญหาสนามหญ้าเปียก - โครงสร้างเครือข่าย
📝 โจทย์: พิจารณาปัจจัยที่ทำให้สนามหญ้าเปียก ได้แก่ สภาพอากาศ (เมฆครึ้ม) ฝนตก และการเปิดระบบรดน้ำ (Sprinkler)

🔗 โครงสร้างเบย์เซียนเน็ตเวิร์ค

มีเมฆครึ้ม (C)
↙ ↘
ฝนตก (R)
เปิด Sprinkler (S)
↘ ↙
สนามหญ้าเปียก (W)

🔍 ความสัมพันธ์เชิงสาเหตุ:

  • เมฆครึ้ม (C) → มีผลต่อ ฝนตก (R) และ การเปิด Sprinkler (S)
  • ฝนตก (R) และ การเปิด Sprinkler (S) → มีผลต่อ สนามหญ้าเปียก (W)
  • C เป็น root node (ไม่มี parent)
  • R และ S เป็น child ของ C
  • W เป็น child ของ R และ S

🧮 ส่วนที่ 4: การคำนวณความน่าจะเป็นร่วมกัน - Python Implementation
📝 โจทย์หลัก: หาความน่าจะเป็นร่วมกันของ "สนามหญ้าเปียก โดยสภาพอากาศไม่มีเมฆครึ้ม, มีฝนตก, และเปิด Sprinkler"
สัญลักษณ์: P(W=T, S=T, R=T, C=F)
# ===== Bayesian Network: ปัญหาสนามหญ้าเปียก ===== import numpy as np import pandas as pd import matplotlib.pyplot as plt class BayesianNetwork: def __init__(self): """สร้างเบย์เซียนเน็ตเวิร์คสำหรับปัญหาสนามหญ้าเปียก""" # P(C) - ความน่าจะเป็นของเมฆครึ้ม self.P_C = { True: 0.5, # มีเมฆครึ้ม False: 0.5 # ไม่มีเมฆครึ้ม } # P(R|C) - ความน่าจะเป็นของฝนตกเมื่อรู้สถานะเมฆ self.P_R_given_C = { (True, True): 0.8, # ฝนตกเมื่อมีเมฆครึ้ม (True, False): 0.2, # ไม่ฝนตกเมื่อมีเมฆครึ้ม (False, True): 0.2, # ฝนตกเมื่อไม่มีเมฆครึ้ม (False, False): 0.8 # ไม่ฝนตกเมื่อไม่มีเมฆครึ้ม } # P(S|C) - ความน่าจะเป็นของ Sprinkler เมื่อรู้สถานะเมฆ self.P_S_given_C = { (True, True): 0.1, # เปิด Sprinkler เมื่อมีเมฆครึ้ม (True, False): 0.9, # ไม่เปิด Sprinkler เมื่อมีเมฆครึ้ม (False, True): 0.5, # เปิด Sprinkler เมื่อไม่มีเมฆครึ้ม (False, False): 0.5 # ไม่เปิด Sprinkler เมื่อไม่มีเมฆครึ้ม } # P(W|R,S) - ความน่าจะเป็นของสนามหญ้าเปียกเมื่อรู้สถานะฝนและ Sprinkler self.P_W_given_R_S = { (True, True, True): 0.99, # เปียกเมื่อฝนตก + Sprinkler (True, True, False): 0.01, # ไม่เปียกเมื่อฝนตก + Sprinkler (True, False, True): 0.80, # เปียกเมื่อฝนตก, ไม่มี Sprinkler (True, False, False): 0.20, # ไม่เปียกเมื่อฝนตก, ไม่มี Sprinkler (False, True, True): 0.90, # เปียกเมื่อไม่ฝนตก, มี Sprinkler (False, True, False): 0.10, # ไม่เปียกเมื่อไม่ฝนตก, มี Sprinkler (False, False, True): 0.00, # เปียกเมื่อไม่ฝนตก, ไม่มี Sprinkler (False, False, False): 1.00 # ไม่เปียกเมื่อไม่ฝนตก, ไม่มี Sprinkler } def calculate_joint_probability(self, C, R, S, W, show_steps=True): """ คำนวณความน่าจะเป็นร่วมกัน P(C, R, S, W) Parameters: - C: สถานะเมฆครึ้ม (True/False) - R: สถานะฝนตก (True/False) - S: สถานะ Sprinkler (True/False) - W: สถานะสนามหญ้าเปียก (True/False) """ if show_steps: print("🔢 การคำนวณความน่าจะเป็นร่วมกัน") print("=" * 50) print(f"เหตุการณ์: C={C}, R={R}, S={S}, W={W}") print() # หาค่าความน่าจะเป็นแต่ละส่วน p_c = self.P_C[C] p_r_given_c = self.P_R_given_C[(C, R)] p_s_given_c = self.P_S_given_C[(C, S)] p_w_given_r_s = self.P_W_given_R_S[(R, S, W)] if show_steps: print("📐 สูตร: P(C,R,S,W) = P(C) × P(R|C) × P(S|C) × P(W|R,S)") print() print("🔍 ค่าที่หาได้จากตาราง CPT:") print(f" P(C={C}) = {p_c}") print(f" P(R={R}|C={C}) = {p_r_given_c}") print(f" P(S={S}|C={C}) = {p_s_given_c}") print(f" P(W={W}|R={R},S={S}) = {p_w_given_r_s}") print() print("💡 การคำนวณ:") print(f" P(C,R,S,W) = {p_c} × {p_r_given_c} × {p_s_given_c} × {p_w_given_r_s}") # คำนวณความน่าจะเป็นร่วม joint_prob = p_c * p_r_given_c * p_s_given_c * p_w_given_r_s if show_steps: print(f" P(C,R,S,W) = {joint_prob:.6f}") print() print(f"✅ คำตอบ: {joint_prob:.6f} หรือ {joint_prob*100:.4f}%") return joint_prob # สร้าง Bayesian Network bn = BayesianNetwork() print("🌟 Bayesian Network: ปัญหาสนามหญ้าเปียก") print("=" * 60) # แก้โจทย์หลัก: P(W=T, S=T, R=T, C=F) print("🎯 โจทย์หลัก: สนามหญ้าเปียก + ไม่มีเมฆครึ้ม + ฝนตก + เปิด Sprinkler") result_main = bn.calculate_joint_probability( C=False, # ไม่มีเมฆครึ้ม R=True, # ฝนตก S=True, # เปิด Sprinkler W=True # สนามหญ้าเปียก ) print("\n" + "="*60) print("🔬 ตัวอย่างเพิ่มเติม") # ตัวอย่างที่ 2: P(W=T, S=F, R=T, C=T) print("\n📝 ตัวอย่างที่ 2: สนามหญ้าเปียก + มีเมฆครึ้ม + ฝนตก + ไม่เปิด Sprinkler") result_2 = bn.calculate_joint_probability( C=True, # มีเมฆครึ้ม R=True, # ฝนตก S=False, # ไม่เปิด Sprinkler W=True # สนามหญ้าเปียก ) # ตัวอย่างที่ 3: P(W=F, S=F, R=F, C=F) print("\n📝 ตัวอย่างที่ 3: สนามหญ้าไม่เปียก + ไม่มีเมฆครึ้ม + ไม่ฝนตก + ไม่เปิด Sprinkler") result_3 = bn.calculate_joint_probability( C=False, # ไม่มีเมฆครึ้ม R=False, # ไม่ฝนตก S=False, # ไม่เปิด Sprinkler W=False # สนามหญ้าไม่เปียก ) print("\n🎓 สรุป: ได้เรียนรู้การคำนวณ Joint Probability ใน Bayesian Network แล้ว!")

💡 แบบฝึกหัด 4.1: คำนวณด้วยตัวเองพร้อมส่งผลการทำงานในแต่ละข้อ

ให้นักศึกษาคำนวณความน่าจะเป็นร่วมกันของสถานการณ์ต่อไปนี้:

  1. สถานการณ์ A: P(W=T, S=T, R=F, C=F) - สนามหญ้าเปียก + ไม่มีเมฆ + ไม่ฝนตก + เปิด Sprinkler
  2. สถานการณ์ B: P(W=F, S=F, R=F, C=T) - สนามหญ้าไม่เปียก + มีเมฆ + ไม่ฝนตก + ไม่เปิด Sprinkler

วิธีทำ: ใช้สูตร P(C,R,S,W) = P(C) × P(R|C) × P(S|C) × P(W|R,S)

🎯 การประยุกต์ใช้ในโลกจริง
📝 สถานการณ์จริง: ระบบตัดสินใจอัตโนมัติสำหรับการรดน้ำสวน
# ===== ระบบตัดสินใจอัตโนมัติสำหรับการรดน้ำสวน ===== class SmartIrrigationSystem: """ระบบรดน้ำอัจฉริยะที่ใช้ Bayesian Network""" def __init__(self, bayesian_network): self.bn = bayesian_network def should_turn_on_sprinkler(self, weather_forecast, current_conditions): """ ตัดสินใจว่าควรเปิดระบบรดน้ำหรือไม่ Parameters: - weather_forecast: {'cloudy': bool, 'rain_probability': float} - current_conditions: {'soil_moisture': float} # 0-1 """ print("🌱 ระบบตัดสินใจการรดน้ำอัจฉริยะ") print("=" * 50) cloudy = weather_forecast['cloudy'] rain_prob = weather_forecast['rain_probability'] soil_moisture = current_conditions['soil_moisture'] print(f"📊 ข้อมูลปัจจุบัน:") print(f" - มีเมฆครึ้ม: {cloudy}") print(f" - โอกาสฝนตก: {rain_prob*100:.1f}%") print(f" - ความชื้นดิน: {soil_moisture*100:.1f}%") # คำนวณความน่าจะเป็นที่ต้องรดน้ำ scenarios = [ (cloudy, True, True), # มีฝน + เปิด sprinkler (cloudy, True, False), # มีฝน + ไม่เปิด sprinkler (cloudy, False, True), # ไม่มีฝน + เปิด sprinkler (cloudy, False, False) # ไม่มีฝน + ไม่เปิด sprinkler ] expected_wetness = 0 for c, r, s in scenarios: # ความน่าจะเป็นของสถานการณ์นี้ p_scenario = (rain_prob if r else (1-rain_prob)) * (0.5 if s else 0.5) # ความน่าจะเป็นที่หญ้าเปียกในสถานการณ์นี้ p_wet = self.bn.P_W_given_R_S[(r, s, True)] expected_wetness += p_scenario * p_wet # ตัดสินใจขึ้นอยู่กับความชื้นดินและความน่าจะเป็นที่หญ้าจะเปียก moisture_threshold = 0.3 # ต้องการความชื้นอย่างน้อย 30% if soil_moisture < moisture_threshold: # ดินแห้ง ต้องพิจารณาเปิด sprinkler if rain_prob < 0.3: # โอกาสฝนต่ำ recommendation = "เปิด Sprinkler" reason = f"ดินแห้ง ({soil_moisture*100:.1f}%) และโอกาสฝนต่ำ ({rain_prob*100:.1f}%)" else: recommendation = "รอดูฝน" reason = f"โอกาสฝนสูง ({rain_prob*100:.1f}%) อาจไม่ต้องรดน้ำ" else: recommendation = "ไม่ต้องเปิด Sprinkler" reason = f"ความชื้นดินเพียงพอ ({soil_moisture*100:.1f}%)" print(f"\n🤖 คำแนะนำของระบบ: {recommendation}") print(f"💡 เหตุผล: {reason}") print(f"📈 ความน่าจะเป็นที่หญ้าจะเปียก: {expected_wetness*100:.1f}%") return recommendation, reason, expected_wetness def test_smart_irrigation(): """ทดสอบระบบรดน้ำอัจฉริยะ""" system = SmartIrrigationSystem(bn) # ทดสอบสถานการณ์ต่างๆ test_cases = [ { "name": "วันแดดจัด", "weather": {"cloudy": False, "rain_probability": 0.1}, "conditions": {"soil_moisture": 0.2} }, { "name": "วันมีเมฆ", "weather": {"cloudy": True, "rain_probability": 0.7}, "conditions": {"soil_moisture": 0.25} }, { "name": "หลังฝนตก", "weather": {"cloudy": True, "rain_probability": 0.9}, "conditions": {"soil_moisture": 0.8} }, { "name": "ดินแห้งมาก", "weather": {"cloudy": False, "rain_probability": 0.2}, "conditions": {"soil_moisture": 0.1} } ] results = [] for test_case in test_cases: print(f"\n🧪 ทดสอบ: {test_case['name']}") print("-" * 40) recommendation, reason, wetness_prob = system.should_turn_on_sprinkler( test_case['weather'], test_case['conditions'] ) results.append({ "สถานการณ์": test_case['name'], "คำแนะนำ": recommendation, "เหตุผล": reason[:50] + "..." if len(reason) > 50 else reason }) # สรุปผลลัพธ์ print("\n📋 สรุปผลการทดสอบ:") df_results = pd.DataFrame(results) display(df_results) # ทดสอบระบบ test_smart_irrigation() print("\n🎯 ประโยชน์ของ Bayesian Network ในการประยุกต์ใช้:") print("✅ ช่วยในการตัดสินใจภายใต้ความไม่แน่นอน") print("✅ สามารถรวมข้อมูลหลายแหล่งได้") print("✅ ให้เหตุผลที่สมเหตุสมผลในการตัดสินใจ") print("✅ ปรับปรุงการตัดสินใจได้เมื่อมีข้อมูลใหม่")

🧠 จุดสำคัญที่ควรจำ:
  1. Bayesian Network แสดงความสัมพันธ์เชิงสาเหตุด้วยโครงสร้างกราฟ
  2. Joint Probability คำนวณจากผลคูณของ conditional probabilities
  3. CPT (Conditional Probability Table) เก็บข้อมูลความน่าจะเป็นแบบมีเงื่อนไข
  4. Monte Carlo Simulation ช่วยตรวจสอบความถูกต้องของการคำนวณ
