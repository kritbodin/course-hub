# ===== Logical Inference Rules ===== class LogicalInference: def __init__(self): self.rules_applied = [] def modus_ponens(self, p_implies_q, p): """ กฎ Modus Ponens: P→Q, P ∴ Q ถ้า P แล้ว Q, และ P เป็นจริง ดังนั้น Q เป็นจริง """ print(f"🔍 ใช้กฎ Modus Ponens:") print(f" กฎ: {p_implies_q}") print(f" ข้อเท็จจริง: {p}") # แยกการประยุกต์กฎ (ง่ายๆ สำหรับตัวอย่าง) if "ถ้า" in p_implies_q and "แล้ว" in p_implies_q: parts = p_implies_q.split("แล้ว") condition = parts[0].replace("ถ้า", "").strip() conclusion = parts[1].strip() if condition == p: print(f" ✅ ข้อสรุป: {conclusion}") self.rules_applied.append("Modus Ponens") return conclusion print(f" ❌ ไม่สามารถใช้กฎได้") return None def modus_tollens(self, p_implies_q, not_q): """ กฎ Modus Tollens: P→Q, ¬Q ∴ ¬P ถ้า P แล้ว Q, และ Q เป็นเท็จ ดังนั้น P เป็นเท็จ """ print(f"🔍 ใช้กฎ Modus Tollens:") print(f" กฎ: {p_implies_q}") print(f" ข้อเท็จจริง: {not_q}") if "ถ้า" in p_implies_q and "แล้ว" in p_implies_q: parts = p_implies_q.split("แล้ว") condition = parts[0].replace("ถ้า", "").strip() conclusion = parts[1].strip() # ตรวจสอบว่า not_q ตรงกับการปฏิเสธของ conclusion if f"ไม่{conclusion}" == not_q or f"ไม่ได้{conclusion}" == not_q: result = f"ไม่{condition}" if not condition.startswith("ไม่") else condition[2:] print(f" ✅ ข้อสรุป: {result}") self.rules_applied.append("Modus Tollens") return result print(f" ❌ ไม่สามารถใช้กฎได้") return None def hypothetical_syllogism(self, p_implies_q, q_implies_r): """ กฎ Hypothetical Syllogism: P→Q, Q→R ∴ P→R ถ้า P แล้ว Q, และ Q แล้ว R ดังนั้น P แล้ว R """ print(f"🔍 ใช้กฎ Hypothetical Syllogism:") print(f" กฎที่ 1: {p_implies_q}") print(f" กฎที่ 2: {q_implies_r}") # แยกส่วนประกอบของกฎ if "ถ้า" in p_implies_q and "แล้ว" in p_implies_q: parts1 = p_implies_q.split("แล้ว") p = parts1[0].replace("ถ้า", "").strip() q = parts1[1].strip() if "ถ้า" in q_implies_r and "แล้ว" in q_implies_r: parts2 = q_implies_r.split("แล้ว") q2 = parts2[0].replace("ถ้า", "").strip() r = parts2[1].strip() # ตรวจสอบว่า Q ในกฎแรกตรงกับ Q ในกฎที่สอง if q == q2: result = f"ถ้า{p}แล้ว{r}" print(f" ✅ ข้อสรุป: {result}") self.rules_applied.append("Hypothetical Syllogism") return result print(f" ❌ ไม่สามารถใช้กฎได้") return None # ===== ตัวอย่างการใช้งาน ===== print("=== การทดสอบกฎการอนุมานเชิงตรรกศาสตร์ ===\n") logic = LogicalInference() print("1️⃣ ทดสอบ Modus Ponens") print("-" * 30) rule1 = "ถ้าฝนตกแล้วถนนเปียก" fact1 = "ฝนตก" result1 = logic.modus_ponens(rule1, fact1) print("\n2️⃣ ทดสอบ Modus Tollens") print("-" * 30) rule2 = "ถ้าเราออกไปแล้วเราจะเจอเพื่อน" fact2 = "ไม่ได้เจอเพื่อน" result2 = logic.modus_tollens(rule2, fact2) print("\n3️⃣ ทดสอบ Hypothetical Syllogism") print("-" * 30) rule3a = "ถ้าฝนตกแล้วถนนเปียก" rule3b = "ถ้าถนนเปียกแล้วขับรถต้องระวัง" result3 = logic.hypothetical_syllogism(rule3a, rule3b) print(f"\n📊 สรุป: ใช้กฎการอนุมาน {len(logic.rules_applied)} ครั้ง") print(f"กฎที่ใช้: {', '.join(logic.rules_applied)}") # ===== Truth Table Generator ===== print("\n" + "="*50) print("=== ตารางค่าความจริง (Truth Table) ===") def generate_truth_table(proposition): """สร้างตารางค่าความจริงสำหรับประพจน์ที่กำหนด""" print(f"\nตารางค่าความจริงสำหรับ: {proposition}") print("-" * 40) # ตัวอย่างง่ายๆ สำหรับ P → Q if "P → Q" in proposition or "P->Q" in proposition: print("P | Q | P→Q") print("--|---|----") combinations = [ (True, True, True), (True, False, False), (False, True, True), (False, False, True) ] for p, q, result in combinations: p_str = "T" if p else "F" q_str = "T" if q else "F" r_str = "T" if result else "F" print(f"{p_str} | {q_str} | {r_str}") # สร้างตารางค่าความจริง generate_truth_table("P → Q")

💡 โจทย์ฝึกหัด 1.1: ตัวอย่างจากสไลด์ - แดดออก

ให้นักศึกษาประยุกต์ใช้ตัวอย่างจากสไลด์เรียน โดยใช้กฎเกี่ยวกับ "แดดออก" ดังนี้:

🌞 กฎจากสไลด์เรียน:
P: แดดออก
Q: เราจะไปชายหาด
R: เราจะเล่นบอลชายหาด
กฎที่ 1: P→Q (ถ้าแดดออก แล้วเราจะไปชายหาด)
กฎที่ 2: Q→R (ถ้าเราจะไปชายหาด แล้วเราจะเล่นบอลชายหาด)

📝 ให้ทำ:

  1. แก้ไขโค้ดในส่วน logic = LogicalInference() โดยเปลี่ยนตัวอย่างให้ใช้:
    • Modus Ponens: กฎ "ถ้าแดดออกแล้วเราจะไปชายหาด" + ข้อเท็จจริง "แดดออก"
    • Modus Tollens: กฎ "ถ้าแดดออกแล้วเราจะไปชายหาด" + ข้อเท็จจริง "ไม่ได้ไปชายหาด"
    • Hypothetical Syllogism: ใช้กฎทั้ง 2 ข้อ (P→Q และ Q→R)
  2. รันโค้ดและบันทึกผลลัพธ์ ของการอนุมานทั้ง 3 แบบ
